Node.js UCI Bridge Script (bridge.js) 
// npm i ws && ENGINE=lc0 PORT=8080 node bridge.js
const { WebSocketServer } = require('ws');
const { spawn } = require('child_process');
const wss = new WebSocketServer({ port: Number(process.env.PORT || 8080) });
wss.on('connection', (ws) => {
  const proc = spawn(process.env.ENGINE || 'lc0', []);
  proc.stdout.on('data', (d) => d.toString().split(/\r?\n/).forEach(l => l && ws.send(l)));
  ws.on('message', (m) => proc.stdin.write(m.toString() + '\n'));
  ws.on('close', () => proc.kill());
});
